Example with Static Scheduling: FIR Filter

Example with Static Scheduling: FIR Filter#

Now consider the following example, which implements an FIR filter without backpressure.

 1// ===== MAIN CODE =============================================================
 2
 3// Length of test inputs
 4const T: u32 = 24
 5// Number of taps
 6const DEPTH: u32 = 3
 7// FIR filter coefficients
 8const COEFFS: Vec[i18, DEPTH] = [3:i18, 5:i18, 7:i18]v
 9// Number of pipeline registers to enable in the DSPs
10const PIPELINE: u32 = 2
11
12accelerator[
13    // Disable the handshake protocol, i.e., use static scheduling
14    no_handshake,
15    // Until we start receiving the real data, the valid bit should be 0.
16    // If the Sirop compiler needs to increase the latency of this stream
17    // (e.g., for latency matching), it must ensure the valid bit remains 0.
18    head(s)=(undefined:i18, false)
19]
20fir = (s: Stm[(i18, bool), T]) =>
21    // ----- Compute "data" part of output -------------------------------------
22    let data =
23        s
24        // Take just the "data" part of the input stream
25        .StmMap( @(data, valid) => data )
26        // Get sliding windows of size 3
27        .StmSlide(3)
28        // Take dot product between each window and the coefficients.
29        // StmMapDot is specifically defined to facilitate the use of the DSPs
30        // in "systolic" mode.
31        .StmMapDot(StmCst(T-(DEPTH-1), COEFFS), PIPELINE)
32        // StmMapDot produces wider outputs for compatibility with DSPs.
33        // Truncate the outputs back to 18 bits.
34        .StmMap( x => truncate18(x) )
35    in
36    // ----- Compute output "valid" bit ----------------------------------------
37    // Consistently set the "head" to false, so that the initial value of each
38    // pipeline register will be false
39    let valid =
40        s
41        // Take just the "valid" bit from the input stream
42        .StmMap( @(data, valid) => valid, /*head=*/false)
43        // Get sliding windows of size 3
44        .StmSlide(3, /*head=*/false)
45        // Output is valid if all elements in the window are valid
46        .StmMap( window => window.VecAll(), /*head=*/false)
47    in
48    // ----- Pair the output "data" with the output "valid" --------------------
49    StmZip(data, valid, /*head=*/(undefined:i18, false))
50
51// ===== TESTING ===============================================================
52
53assert {
54    // Square wave with amplitude 42 and period 8
55    s = [
56        // Maybe the first few cycles will have invalid data
57        (undefined:i18, false),
58        (undefined:i18, false)
59    ]s ++ [
60        (-42:i18, true), (-42:i18, true), (-42:i18, true), (-42:i18, true),
61        (+42:i18, true), (+42:i18, true), (+42:i18, true), (+42:i18, true),
62        (-42:i18, true), (-42:i18, true), (-42:i18, true), (-42:i18, true),
63        (+42:i18, true), (+42:i18, true), (+42:i18, true), (+42:i18, false),
64        (-42:i18, true), (-42:i18, true), (-42:i18, true), (-42:i18, true),
65        (+42:i18, true), (+42:i18, true), (+42:i18, true), (+42:i18, true)
66    ]s
67}
68yields [
69    (-630:i18, true), (-630:i18, true),  (-42:i18, true),  (+378:i18, true),
70    (+630:i18, true), (+630:i18, true),  (+42:i18, true),  (-378:i18, true),
71    (-630:i18, true), (-630:i18, true),  (-42:i18, true),  (+378:i18, true),
72    (+630:i18, true), (+630:i18, false), (+42:i18, false), (-378:i18, false),
73    (-630:i18, true), (-630:i18, true),  (-42:i18, true),  (+378:i18, true),
74    (+630:i18, true), (+630:i18, true)
75]s
76// The FIR filter will have some latency N.
77// For the first N clock cycles, the output "valid" bit must be 0.
78with prefix /* which satisfies the predicate... */ @(_, valid) => valid == false

Test, compile, and simulate the design using the following command.

Note

To change the compilation target, use the --out:vhdl:family and --out:vhdl:device flags. Run sirop --help for more details.

$ sirop fir.sirop --out:test --out:vhdl vhdl_project_dir/ --out:vhdl:run-sim
