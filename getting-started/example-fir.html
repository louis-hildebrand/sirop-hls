Example with Static Scheduling: FIR Filter

Example with Static Scheduling: FIR Filter#

Now consider the following example, which implements an FIR filter without backpressure.

  1// ===== MAIN CODE =============================================================
  2
  3// Length of test inputs
  4const T: u32 = 24
  5// Number of taps.
  6// Even numbers are more resource-efficient, since each DSP can perform two
  7// multiplications in systolic mode.
  8const DEPTH: u32 = 4
  9// Number of pipeline registers to enable in the DSPs.
 10// Enabling more of these tends to increases the maximum clock frequency,
 11// but it also increases the latency.
 12// In some cases, enabling more registers may also increase resource usage.
 13const PIPELINE: u32 = 2
 14
 15accelerator[
 16    // Disable the handshake protocol, i.e., use static scheduling
 17    no_handshake,
 18    // Until we start receiving the real data, the valid bit should be 0.
 19    // If the Sirop compiler needs to increase the latency of this stream
 20    // (e.g., for latency matching), it must ensure the valid bit remains 0.
 21    head(s)=(undefined:i18, false)
 22]
 23fir = (s: Stm[(i18, bool), T]) => (weights: Stm[Vec[i18, DEPTH], T]) =>
 24    // ----- Compute "data" part of output -------------------------------------
 25    let data =
 26        s
 27        // Take just the "data" part of the input stream
 28        .StmMap( @(data, valid) => data )
 29        // Get sliding windows
 30        .StmSlide(DEPTH)
 31        // Take dot product between each window and the coefficients.
 32        // StmMapDotCascaded is specifically defined to facilitate the use of
 33        // the DSPs in "systolic" mode.
 34        .StmCascade()
 35        .StmMapDotCascaded(weights, PIPELINE)
 36        // StmMapDotCascaded produces wider outputs for compatibility with DSPs.
 37        // Truncate the outputs back to 18 bits.
 38        .StmMap( x => truncate18(x) )
 39    in
 40    // ----- Compute output "valid" bit ----------------------------------------
 41    // Consistently set the "head" to false, so that the initial value of each
 42    // pipeline register will be false
 43    let valid =
 44        s
 45        // Take just the "valid" bit from the input stream
 46        .StmMap( @(data, valid) => valid, /*head=*/false)
 47        // Get sliding windows
 48        .StmSlide(DEPTH, /*head=*/false)
 49        // Output is valid if all elements in the window are valid
 50        .StmMap( window => window.VecAll(), /*head=*/false)
 51    in
 52    // ----- Pair the output "data" with the output "valid" --------------------
 53    StmZip(data, valid, /*head=*/(undefined:i18, false))
 54
 55// ===== TESTING ===============================================================
 56
 57// FIR filter coefficients
 58const COEFFS: Vec[i18, 4] = [3:i18, 5:i18, 7:i18, 9:i18]v
 59
 60const UNDEF: i18 = undefined:i18
 61
 62assert {
 63    // Square wave with amplitude 42 and period 8
 64    s = [
 65        // Maybe the first few cycles will have invalid data
 66        (undefined:i18, false)
 67    ]s ++ [
 68        (-42:i18, true), (-42:i18, true), (-42:i18, true), (-42:i18, true),
 69        (+42:i18, true), (+42:i18, true), (+42:i18, true), (+42:i18, true),
 70        (-42:i18, true), (-42:i18, true), (-42:i18, true), (-42:i18, true),
 71        (+42:i18, true), (+42:i18, true), (+42:i18, true), (+42:i18, false),
 72        (-42:i18, true), (-42:i18, true), (-42:i18, true), (-42:i18, true),
 73        (+42:i18, true), (+42:i18, true), (+42:i18, true), (+42:i18, true)
 74    ]s,
 75    weights = StmCst(T, COEFFS)
 76}
 77yields [
 78    (-1008:i18, true),  (-252:i18, true),   (+336:i18, true),   (+756:i18, true),
 79    (+1008:i18, true),  (+252:i18, true),   (-336:i18, true),   (-756:i18, true),
 80    (-1008:i18, true),  (-252:i18, true),   (+336:i18, true),   (+756:i18, true),
 81    (+1008:i18, false), (+252:i18, false),  (-336:i18, false),  (-756:i18, false),
 82    (-1008:i18, true),  (-252:i18, true),   (+336:i18, true),   (+756:i18, true),
 83    (+1008:i18, true)
 84]s
 85// The FIR filter will have some latency N.
 86// For the first N clock cycles, the output "valid" bit must be 0.
 87with prefix /* which satisfies the predicate... */ @(_, valid) => valid == false
 88
 89assert {
 90    s = [
 91        (101:i18, true), (102:i18, true), (103:i18, true), (104:i18, true),
 92        (105:i18, true), (106:i18, true), (107:i18, true), (108:i18, true),
 93        (109:i18, true), (110:i18, true), (111:i18, true), (112:i18, true),
 94        (113:i18, true), (114:i18, true), (115:i18, true), (116:i18, true),
 95        (117:i18, true), (118:i18, true), (119:i18, true), (120:i18, true),
 96        (121:i18, true), (122:i18, true), (123:i18, true), (124:i18, true)
 97    ]s,
 98    // Note that the weights must be provided in a "cascaded" manner, i.e.,
 99    // each element in the vector is delayed by one clock cycle compared to
100    // the previous element.
101    // Therefore, the first window of data ([101, 102, 103, 104]) will be
102    // multiplied by [1, 1, 1, 1] in this test case.
103    weights = [
104        [ 1:i18,  UNDEF,  UNDEF,  UNDEF]v,
105        [ 2:i18,  1:i18,  UNDEF,  UNDEF]v,
106        [ 3:i18,  2:i18,  1:i18,  UNDEF]v,
107        [ 4:i18,  3:i18,  2:i18,  1:i18]v,
108        [ 5:i18,  4:i18,  3:i18,  2:i18]v,
109        [ 6:i18,  5:i18,  4:i18,  3:i18]v,
110        [ 7:i18,  6:i18,  5:i18,  4:i18]v,
111        [ 8:i18,  7:i18,  6:i18,  5:i18]v,
112        [ 9:i18,  8:i18,  7:i18,  6:i18]v,
113        [10:i18,  9:i18,  8:i18,  7:i18]v,
114        [11:i18, 10:i18,  9:i18,  8:i18]v,
115        [12:i18, 11:i18, 10:i18,  9:i18]v,
116        [13:i18, 12:i18, 11:i18, 10:i18]v,
117        [14:i18, 13:i18, 12:i18, 11:i18]v,
118        [15:i18, 14:i18, 13:i18, 12:i18]v,
119        [16:i18, 15:i18, 14:i18, 13:i18]v,
120        [17:i18, 16:i18, 15:i18, 14:i18]v,
121        [18:i18, 17:i18, 16:i18, 15:i18]v,
122        [19:i18, 18:i18, 17:i18, 16:i18]v,
123        [20:i18, 19:i18, 18:i18, 17:i18]v,
124        [21:i18, 20:i18, 19:i18, 18:i18]v,
125        [ UNDEF, 21:i18, 20:i18, 19:i18]v,
126        [ UNDEF,  UNDEF, 21:i18, 20:i18]v,
127        [ UNDEF,  UNDEF,  UNDEF, 21:i18]v
128    ]s
129}
130yields [
131    ( 1:i18 * (101:i18 + 102:i18 + 103:i18 + 104:i18), true),
132    ( 2:i18 * (102:i18 + 103:i18 + 104:i18 + 105:i18), true),
133    ( 3:i18 * (103:i18 + 104:i18 + 105:i18 + 106:i18), true),
134    ( 4:i18 * (104:i18 + 105:i18 + 106:i18 + 107:i18), true),
135    ( 5:i18 * (105:i18 + 106:i18 + 107:i18 + 108:i18), true),
136    ( 6:i18 * (106:i18 + 107:i18 + 108:i18 + 109:i18), true),
137    ( 7:i18 * (107:i18 + 108:i18 + 109:i18 + 110:i18), true),
138    ( 8:i18 * (108:i18 + 109:i18 + 110:i18 + 111:i18), true),
139    ( 9:i18 * (109:i18 + 110:i18 + 111:i18 + 112:i18), true),
140    (10:i18 * (110:i18 + 111:i18 + 112:i18 + 113:i18), true),
141    (11:i18 * (111:i18 + 112:i18 + 113:i18 + 114:i18), true),
142    (12:i18 * (112:i18 + 113:i18 + 114:i18 + 115:i18), true),
143    (13:i18 * (113:i18 + 114:i18 + 115:i18 + 116:i18), true),
144    (14:i18 * (114:i18 + 115:i18 + 116:i18 + 117:i18), true),
145    (15:i18 * (115:i18 + 116:i18 + 117:i18 + 118:i18), true),
146    (16:i18 * (116:i18 + 117:i18 + 118:i18 + 119:i18), true),
147    (17:i18 * (117:i18 + 118:i18 + 119:i18 + 120:i18), true),
148    (18:i18 * (118:i18 + 119:i18 + 120:i18 + 121:i18), true),
149    (19:i18 * (119:i18 + 120:i18 + 121:i18 + 122:i18), true),
150    (20:i18 * (120:i18 + 121:i18 + 122:i18 + 123:i18), true),

152]s
153with prefix @(_, valid) => valid == false

Note

To change the compilation target, use the --out:vhdl:family and --out:vhdl:device flags. Run sirop --help for more details.
